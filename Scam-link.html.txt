<script>

const no=document.getElementById("no");
const yes=document.getElementById("yes");
const msg=document.getElementById("message");

const messages=[
"Nice try 😂",
"Nope 😌",
"Think again 👀",
"You almost got it 😭",
"Wrong answer 😂",
"Try the other button ❤️",
"The universe disagrees 🌎",
"Destiny says YES 😏"
];

let scale=1;

function moveButton(){

const x=Math.random()*220;
const y=Math.random()*90;

no.style.left=x+"px";
no.style.top=y+"px";

scale+=0.08;

yes.style.transform=`scale(${scale})`;

msg.innerHTML=messages[Math.floor(Math.random()*messages.length)];

}

no.addEventListener("mouseover",moveButton);
no.addEventListener("touchstart",function(e){
e.preventDefault();
moveButton();
});

yes.onclick=function(){

document.body.innerHTML=`
<div style="
display:flex;
justify-content:center;
align-items:center;
flex-direction:column;
height:100vh;
background:#ffeff6;
font-family:Arial;
text-align:center;
">
<h1 style="font-size:60px;">🎉❤️🎉</h1>
<h1>YAYYYYY!!</h1>
<h2>You just made me the happiest person ❤️</h2>
<p>I promise I'll make this worth your click 😌</p>
</div>
`;

for(let i=0;i<150;i++){

let c=document.createElement("div");
c.className="confetti";
c.innerHTML=["🎉","❤️","💖","✨"][Math.floor(Math.random()*4)];

c.style.left=Math.random()*100+"vw";
c.style.animationDuration=(Math.random()*2+1)+"s";

document.body.appendChild(c);

}

}

document.addEventListener("mousemove",e=>{

let h=document.createElement("div");
h.className="heart";
h.innerHTML="❤️";

h.style.left=e.clientX+"px";
h.style.top=e.clientY+"px";

document.body.appendChild(h);

setTimeout(()=>h.remove(),900);

});
